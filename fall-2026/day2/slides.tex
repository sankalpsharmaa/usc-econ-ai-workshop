    \begin{tabular}{@{}p{0.20\textwidth} p{0.24\textwidth} p{0.48\textwidth}@{}}

def test_boot_se_matches_ols():
    rng = np.random.default_rng(42)
    np.testing.assert_allclose(
        boot_se(X, y, rng), ols_se(X, y),
        rtol=0.05)
\end{lstlisting}

        \column{0.40\textwidth}
