  \vspace{0.4cm}
  \begin{itemize}
    \setlength\itemsep{0.45em}
    \item \textbf{Closed feedback.} Each tool call returns a result (a file, an error, a test failure) and that result becomes the next prompt
    \item \textbf{Goal-level, not step-level.} You say \textit{``make the tests pass.''} The model picks the steps. 
    \item \textbf{Stops at the goal, not at the script.} The loop ends when the model decides the goal is met.
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{Before you Claude, get on git}
  \hypertarget{main:git}{}
  \appbtn{worktrees}{git worktrees}
  \vspace{0.2cm}
  Version control is essential!
  \vspace{0.2cm}
  \begin{lstlisting}[style=shell]
git init && gh repo create my-paper --private --source=.
echo "data/raw/"     >> .gitignore   # never commit raw data
echo "*.dta *.parquet *.log" >> .gitignore
echo ".env"          >> .gitignore   # never commit secrets
git add code/ paper/ CLAUDE.md && git commit -m "init"
  \end{lstlisting}
  \vspace{0.3cm}
  \textbf{Three habits that matter once Claude is editing your files:}
  \begin{itemize}
    \setlength\itemsep{0.3em}
    \item \textbf{One spec, one branch.} \texttt{git checkout -b feat/event-study}. Bad runs throw away cleanly.
    \item \textbf{\texttt{git diff} before approving.} 
    \item \textbf{Worktrees for parallel sessions.} \texttt{git worktree add ../paper-did feat/did}, then \texttt{cd} in and run \texttt{claude}. Three Claudes, three modules
  \end{itemize}
  \vspace{0.2cm}
\end{frame}

\begin{frame}[fragile]{The five surfaces every config touches}

\begin{frame}[fragile]{\texttt{CLAUDE.md} }
  \hypertarget{main:claudemd}{}
  \appbtn{path-rules}{path-scoped rules}
  \vspace{0.2cm}
  \texttt{CLAUDE.md} loads on every session. It is the single highest-leverage file in your project.
  \begin{lstlisting}[style=shell]
# CLAUDE.md (project root)
## Project: 1-line topic. Stack. Master entry point.
## Conventions: FE engine = reghdfe / fixest / pyfixest.
                Cluster SE level = district. Seed = 20260415.
                Tables = booktabs + esttab/modelsummary.
## Forbidden: pdflatex (use xelatex). pandas for >5GB.
              Plain TWFE for staggered designs.
## Replication: make all reproduces every table from data/raw/.
  \end{lstlisting}
  \vspace{0.2cm}
  \textbf{Two-tier setup:}
  \begin{itemize}
    \setlength\itemsep{0.25em}
    \item \texttt{\textasciitilde/.claude/CLAUDE.md} (you, all projects): writing voice, banned phrases, default models.
    \item \texttt{./CLAUDE.md} (project, committed): conventions, file layout, what never to touch.
  \end{itemize}
  \vspace{0.1cm}
  \footnotesize Test for every line: \emph{``Would removing this cause Claude to make a mistake?''} If not, cut.
\end{frame}

\begin{frame}[fragile]{Custom commands: \texttt{/prompt} turns dictation into a real prompt}
  \vspace{0.2cm}
  A command lives at \texttt{\textasciitilde/.claude/commands/prompt.md}. The command body is itself a prompt that runs when you type \texttt{/prompt}.
  \begin{lstlisting}[style=shell]
~/.claude/commands/prompt.md
~/.claude/commands/learn.md            # extract lessons from this session
~/.claude/commands/review.md           # review unpushed changes
~/.claude/commands/today.md            # session progress summary
~/.claude/commands/scrape.md           # build a scraper for a portal
~/.claude/commands/interview.md        # flesh out a spec via Q&A
  \end{lstlisting}
  \vspace{0.2cm}
  \textbf{What \texttt{/prompt} does:} I dictate a sloppy half-sentence. The command (a) parses intent, (b) calibrates depth (light/standard/deep), (c) reformats into role + task + constraints + output format, (d) shows me the formatted version, (e) executes.
  \\
  \vspace{0.2cm}
  \footnotesize Net effect: the bottleneck moves from ``what do I type'' to ``what do I want.''
\end{frame}

\begin{frame}[fragile]{Skills: bundles of expertise the agent loads on demand}
  \hypertarget{main:skills}{}
  \appbtn{learning}{continuous learning}
  \vspace{0.2cm}
  A skill is a folder with a \texttt{SKILL.md} (frontmatter + instructions) plus reference files.
  \\
  Claude reads the frontmatter, decides if the skill is relevant, then pulls in the body.
  \begin{lstlisting}[style=shell]
~/.claude/skills/econometrics/         # OLS, IV, GMM, panel, treatment eval
~/.claude/skills/research-pipeline-qa/ # raw data to paper, end-to-end audits
~/.claude/skills/hpc/                  # Slurm scripts, USC CARC, modules
~/.claude/skills/scientific-visualization/  # journal-ready matplotlib/seaborn
  \end{lstlisting}
  \vspace{0.2cm}
  \textbf{Skills $>>$ CLAUDE.md:} Skills load only when triggered, so context stays cheap.
\end{frame}

\begin{frame}[fragile]{Hooks: deterministic shell commands that fire at lifecycle events}
  \vspace{0.2cm}
  \textbf{Hooks guarantee behavior (prompts only suggest it)} 
  \\

  \begin{lstlisting}[style=shell]
"PreToolUse":  pre-bash         # block rm -rf, git push --force, secrets
               pre-edit          # block edits to data/raw/, .env, IRB folders
               pre-write         # check no PII in new files
"PostToolUse": post-python-edit  # auto-format with ruff
               post-latex-edit   # try compiling, push errors back into context
               post-tool-error   # capture errors for the [LEARN] memory
"SessionStart": auto-commit memory/ folder, sync configs across machines
"UserPromptSubmit": skill-router.py  # match prompt against regex, load skill
"Stop": claude-sync push        # back up the session
  \end{lstlisting}
  \vspace{0.1cm}
\end{frame}

\begin{frame}[fragile]{Agents: a fresh context window with a custom system prompt}

~/.claude/agents/python-pro.md            # refactors, async, performance
~/.claude/agents/search-papers.md         # semantic search over Zotero
  \end{lstlisting}
  \vspace{0.2cm}

\end{frame}

\begin{frame}[fragile]{Opus as the advisor, Sonnet as the worker}
  \hypertarget{main:models}{}
  \appbtn{models}{model details}
  \vspace{0.2cm}

  \begin{itemize}
    \setlength\itemsep{0.3em}
    \item One git repo, one place to point Claude.
    \item \texttt{brainstorm/} and \texttt{plan/} are version-controlled, so you can blame your past self.
    \item \texttt{source/} is gitignored; \texttt{code/} and \texttt{paper/} are committed.
    \item Path-scoped rules apply naturally: writing rules in \texttt{paper/}, code rules in \texttt{code/}.
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{Path-scoped rules}
  \hypertarget{app:path-rules}{}
  \backbtn{claudemd}{back}
  \vspace{0.2cm}
  Instead of one giant \texttt{CLAUDE.md}, split conventions by file type and let glob patterns do the routing:
  \begin{lstlisting}[style=shell]
~/.claude/rules/
  global.md       # applies always
  stata.md        # applies-to: **/*.do, **/*.ado
  python.md       # applies-to: **/*.py, **/*.ipynb
  r.md            # applies-to: **/*.R, **/*.Rmd
  data-work.md    # applies-to: **/data/**, **/build/**
  research.md     # applies-to: **/analysis/**, **/*.tex
  gis.md          # applies-to: **/*.shp, **/*.geojson
  writing.md      # applies-to: **/*.md, **/docs/**
  \end{lstlisting}
  \vspace{0.2cm}
  \textbf{Why this beats one big file:}
  \begin{itemize}
    \setlength\itemsep{0.25em}
    \item Stata-specific gotchas do not pollute Python sessions.
    \item Each file stays under the canonical 200-line limit.
    \item Sub-projects can override at the directory level.
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{Continuous learning}
  \hypertarget{app:learning}{}
  \backbtn{skills}{back}
  \vspace{0.2cm}
  After every correction or surprising success, write a one-line memory. Future sessions read it on startup.
  \begin{lstlisting}[style=shell]
~/.claude/memory/
  MEMORY.md           # index, max 200 lines, one line per entry
  user_profile.md     # role, tools, knowledge
  feedback_*.md       # corrections you do not want to repeat
  project_*.md        # current state of in-flight work
  \end{lstlisting}
  \vspace{0.2cm}
  \textbf{The rule:} after the second correction on the same issue, write a feedback memory. After the second time an unusual approach worked, write one too.
  \vspace{0.2cm}
  \begin{itemize}
    \setlength\itemsep{0.25em}
    \item \emph{``PLFS weights need adjusted multipliers for round 7+.''}
    \item \emph{``For staggered DiD, default to Callaway-Sant'Anna.''}
    \item \emph{``Never use stargazer. Always modelsummary or esttab.''}
  \end{itemize}
  \vspace{0.1cm}
  \footnotesize Mistake rate falls fast once you have 20 entries. This is the closest thing to actual learning the agent does.
\end{frame}

\begin{frame}[fragile]{Git worktrees: three Claudes on three branches at once}
  \hypertarget{app:worktrees}{}
  \backbtn{git}{back}
  \vspace{0.2cm}
  \begin{lstlisting}[style=shell]
# git creates a sibling checkout on a new branch
git worktree add ../paper-did feat/staggered-did
cd ../paper-did && claude   # fresh Claude session, isolated tree

# repeat for each parallel module
git worktree add ../paper-cleaning feat/clean-rounds

# list, then prune when done
git worktree list
git worktree remove ../paper-did
  \end{lstlisting}
  \vspace{0.3cm}
  \textbf{The pattern:}
  \begin{itemize}
    \setlength\itemsep{0.3em}
    \item Three terminal panes. Three branches. Three Claude sessions. Zero merge conflicts during the session because each lives in its own checkout.
    \item Scope by \emph{module}, not by task. (\texttt{cleaning/}, \texttt{analysis/}, \texttt{paper/}.)
    \item Practical ceiling: 3--5 parallel sessions. Beyond that you cannot keep up with the diffs.
    \item Cost: a 3-agent hour costs about a single-agent day. Use deliberately.
  \end{itemize}
\end{frame}
